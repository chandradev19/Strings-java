// Defanging an IP Address (LC - 1108 Loop)

class Solution {
    public String defangIPaddr(String address) {
        String result = "";
        for (int i = 0; i < address.length(); i++) {
            char current = address.charAt(i);
            if (current == '.') {
                result = result + "[.]";
            } else {
                result = result + current;
            }
        }
        return result;
    }
}
